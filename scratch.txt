df
df.columns

df.drop_duplicates(inplace=True)
# Standardize column names
df.columns = df.columns.str.strip()

print(df.head())
print(df.shape)
print(df.columns)
df.describe()
# 5. Identify outliers using IQR method

# Select numerical columns
numeric_cols = df.select_dtypes(include=[np.number])
Q1 = numeric_cols.quantile(0.25)
Q3 = numeric_cols.quantile(0.75)
IQR = Q3 - Q1
outliers = ((numeric_cols < (Q1 - 1.5 * IQR)) |
            (numeric_cols > (Q3 + 1.5 * IQR)))
print("Outliers using IQR:")
print(outliers.sum())
# 6. Identify outliers using Z-score method

z_scores = np.abs(stats.zscore(numeric_cols))
print("Outliers using Z-score:")
print((z_scores > 3).sum(axis=0))
# 7. Apply data transformation

# Log transformation
df_log = numeric_cols.apply(lambda x: np.log1p(x))

# Normalization using Min-Max Scaling
scaler = MinMaxScaler()
df_normalized = pd.DataFrame(scaler.fit_transform(numeric_cols),
                             columns=numeric_cols.columns)
# 8. Display transformed data
print("Log Transformed Data:")
print(df_log.head())

print("Normalized Data:")
print(df_normalized.head())

text = "Natural Language Processing is amazing and helps in analyzing text data easily."
tokenize_text = sent_tokenize(text)
# Breaking sentence into words/tokens.
print(tokenize_text)
tokens = word_tokenize(text) #sentence.split(" ")
print("Tokens:", tokens)

# POS Tagging
pos = pos_tag(tokens)
# Assigns grammatical category to each word.
print("POS Tags:", pos)

# Tag	Meaning
# NN	Noun
# VB	Verb
# JJ	Adjective
# RB	Adverb
# Stopword Removal
filtered = [word for word in tokens if word.lower() not in stopwords.words('english')]
print("After Stopword Removal:", filtered)
# Stemming
ps = PorterStemmer()
stemmed = [ps.stem(word) for word in filtered]
print("Stemmed Words:", stemmed)

# Converts words to root form.
# Lemmatization
lm = WordNetLemmatizer()
lemmatized = [lm.lemmatize(word) for word in filtered]
print("Lemmatized Words:", lemmatized)
